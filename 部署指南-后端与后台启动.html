三、app.js 正确代码模板

以下是 /home/project/server/src/app.js 的完整正确结构：

require('dotenv').config(); const express = require('express'); const cors = require('cors'); const path = require('path'); // 引入路由 const authRoutes = require('./routes/auth'); const taskRoutes = require('./routes/tasks'); // ... 其他路由 ... const app = express(); const PORT = process.env.PORT || 3000; // 中间件 app.use(cors()); app.use(express.json({ limit: '10mb' })); app.use(express.urlencoded({ extended: true })); // 上传文件静态服务 app.use('/uploads', express.static(path.join(__dirname, '../uploads'))); // ========== ❶ 通用 API 路由（必须在前） ========== app.use('/api/auth', authRoutes); app.use('/api/tasks', taskRoutes); app.use('/api/learning-records', learningRecordRoutes); app.use('/api/exam-records', examRecordRoutes); app.use('/api/reports', reportRoutes); // ========== ❷ 管理后台 API 路由 ========== app.use('/api/admin/organizations', adminOrgRoutes); app.use('/api/admin/users', adminUserRoutes); app.use('/api/admin/contents', adminContentRoutes); app.use('/api/admin/tasks', adminTaskRoutes); app.use('/api/admin/questions', adminQuestionRoutes); app.use('/api/admin/statistics', adminStatisticsRoutes); // ========== ❸ 托管静态文件 + SPA 兜底（最后！）========== app.use(express.static(path.join(__dirname, '../../admin/dist'))); app.get('*', (req, res) => { res.sendFile(path.join(__dirname, '../../admin/dist/index.html')); }); // ========== ❹ 全局错误处理 ========== app.use((err, req, res, next) => { console.error('Unhandled error:', err); res.status(500).json({ code: 500, message: '服务器内部错误', data: null }); }); app.listen(PORT, () => { console.log(`服务已启动，端口: ${PORT}`); });

四、踩过的坑 & 经验总结

🔴 坑 1：PM2 多进程抢端口

现象：改了代码重启后还是旧的 404 响应

原因：之前用不同名字（如 party-school 和 party-school-server）启动了多个 PM2 进程，旧进程占着 3000 端口，新进程的代码根本没生效。

解决：

pm2 list # 查看所有进程，确认只有一个 pm2 delete all # 全部删掉 pm2 start src/app.js --name party-school-server pm2 save

🔴 坑 2：静态文件托管顺序错误

现象：curl http://localhost:3000/api/users/me 返回 HTML 而不是 JSON

原因：app.get('*', ...) 写在了 API 路由前面，* 匹配所有请求，把 API 请求全拦截成 HTML 了。

解决：静态文件托管和 * 兜底必须放在所有 /api/ 路由之后。

🔴 坑 3：pm2 restart 没生效

现象：pm2 restart 后代码还是旧的

原因：罕见情况下 restart 不会重新加载文件

解决：用 pm2 delete 再 pm2 start 彻底重建进程。

✅ 核心原则：Express 路由匹配是从上到下的，
  • API 路由必须在上面 → /api/xxx
  • 静态文件托管必须在下面 → * 兜底
  • 顺序一旦颠倒，所有 API 都会被拦截成 HTML

初次部署完成后，app.js 不需要再改。以后更新代码，只需上传文件后执行对应命令即可。
